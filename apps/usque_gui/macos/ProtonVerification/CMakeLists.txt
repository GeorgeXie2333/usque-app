cmake_minimum_required(VERSION 3.21)
project(UsqueProtonVerification LANGUAGES C CXX OBJCXX)
if(NOT APPLE OR NOT DEFINED CEF_ROOT OR NOT USE_SANDBOX)
  message(FATAL_ERROR "A macOS CEF distribution with USE_SANDBOX=ON is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_executable(UsqueProtonVerification main.mm)
add_executable(UsqueProtonVerificationHelper helper.cc)
foreach(target UsqueProtonVerification UsqueProtonVerificationHelper)
  target_include_directories(${target} PRIVATE "${CEF_ROOT}")
  target_compile_definitions(${target} PRIVATE CEF_USE_SANDBOX)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-parameter)
  target_link_libraries(${target} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
endforeach()
target_compile_options(UsqueProtonVerification PRIVATE -fobjc-arc)
target_link_libraries(UsqueProtonVerification PRIVATE "-framework Security")
